Execução do Bubble Sort

Demonstração do Bubble Sort

def bubble_sort(lista):
        n = len(lista)

        for i in range(n):
            for j in range(0, n - i - 1):
                if lista[j] > lista[j + 1]:
                    lista[j], lista[j + 1] = (
                        lista[j + 1],
                        lista[j]
                    )

        return lista

def insertion_sort(lista):

        chave = lista[i]
        j = i - 1

        while j >= 0 and lista[j] > chave:
            lista[j + 1] = lista[j]
            j -= 1

        lista[j + 1] = chave

    return lista
